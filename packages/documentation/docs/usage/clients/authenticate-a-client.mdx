## Overview

If supported, providers can sign some typed authentication data.

:::info

When verifying the signature, it **SHOULD** be signed using the [ARC-0060](https://arc.algorand.foundation/ARCs/arc-0060) specification. The utility method [`verifyTypedData`](../../api-reference/utilities#verifytypeddataoptions) can be used to verify the signature.

:::

## Authenticate a client

<Tabs
  defaultValue="javascript"
  values={[
    { label: 'Javascript', value: 'javascript' },
    { label: 'TypeScript', value: 'typescript' },
  ]}>
  <TabItem value="javascript">

  ```js
  const { verifyTypedData } = require('@agoralabs-sh/avm-web-provider');
  const { decodeAddress } = require('algosdk');

  const authenticationData = await window.crypto.subtle.digest(
    'SHA-256',
    new TextEncoder().encode(window.location.hostname),
  );
  const data = {
    hello: 'humie',
  };

  client.onAuthenticate(({ error, result }) => {
    if (error) {
      console.error('error:', error);

      return;
    }

    console.log(result);
    /*
    {
      signature: 'gqNzaWfEQ...',
      signer: 'P3AIQVDJ2CTH54KSJE63YWB7IZGS4W4JGC53I6GK72BGZ5BXO2B2PS4M4U',
    }
    */

    if (
      verifyTypedData({
        authenticationData,
        data,
        publicKey: decodeAddress('P3AIQVDJ2CTH54KSJE63YWB7IZGS4W4JGC53I6GK72BGZ5BXO2B2PS4M4U').publicKey,
        scope: 1, // AUTH
        signature: new TextEncoder().encode(window.bota(result.signature)),
      })
    ) {
      console.log('authentication successful');
    }
  });

  // send an authentication request
  client.authenticate({
    params: {
      authenticationData: window.bota(new TextDecoder.decode(authenticationData)),
      data: window.btoa(JSON.stringify(data)),
    },
    vcic: 'q5dQ...',
  });
  ```
